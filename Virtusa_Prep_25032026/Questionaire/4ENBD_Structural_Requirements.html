"""
transaction_etl.py

Purpose  : Incremental ETL pipeline for Emirates NBD core banking transactions.
           Extracts from Oracle source via JDBC, validates, transforms, and
           persists to a Delta Lake Silver layer with dead-letter quarantine routing.

Features :
    - Watermark-based incremental extraction (no full reloads)
    - Dead-letter quarantine routing for invalid records
    - Data quality validation at Bronze → Silver boundary
    - SCD-safe deduplication using window functions
    - Broadcast-join enrichment with branch master data
    - Adaptive Query Execution (AQE) enabled for auto-optimization
    - All external credentials sourced from Azure Key Vault via Databricks secrets

Author   : Sudheer Kumar Musalimadugu
Version  : 1.0.0
Created  : 2024-01-15
"""

# ── Standard library ────────────────────────────────────────────────────
import logging
from datetime import datetime
from typing import Dict, List, Optional, Tuple

# ── Third-party ─────────────────────────────────────────────────────────
from pyspark.sql import DataFrame, SparkSession
from pyspark.sql.functions import (
    array,
    col,
    concat_ws,
    current_timestamp,
    lit,
    max as spark_max,
    row_number,
    size,
    when,
    filter as array_filter,
)
from pyspark.sql.window import Window

# ── Internal / project ──────────────────────────────────────────────────
# (Add internal utility imports here if applicable, e.g. from utils.secrets)

# Module-level logger — each instance configures handlers in _setup_logging()
logger = logging.getLogger(__name__)


class CustomerTransactionETL:
    """
    End-to-end incremental ETL pipeline for ENBD customer transaction data.

    Orchestrates the full pipeline from Oracle core-banking extraction through
    Delta Lake Silver persistence. Implements a strict separation of concerns:
    configuration setup, data loading, processing, enrichment, and output are
    each isolated into dedicated private methods, making the pipeline testable,
    auditable, and maintainable without monolithic side effects.

    Attributes:
        spark        (SparkSession) : Active Spark session for all operations.
        config       (Dict[str, str]): Pipeline configuration parameters including
                                       paths, connection settings, and thresholds.
        run_date     (str)           : ISO date string for the current pipeline run
                                       (YYYY-MM-DD), used in audit columns.
        _jdbc_props  (Dict[str, str]): Lazily initialized JDBC connection properties.

    Example:
        config = {
            "oracle_jdbc_url": "jdbc:oracle:thin:@host:1521/ENBD_PROD",
            "db_user"        : "etl_service",
            "db_password"    : "...",            # from Key Vault in production
            "watermark_path" : "/mnt/meta/watermarks",
            "silver_path"    : "/mnt/silver/transactions",
            "quarantine_path": "/mnt/quarantine/transactions",
            "branch_table"   : "enbd.dim_branch",
        }
        etl = CustomerTransactionETL(spark, config)
        etl.run_etl_pipeline()
    """

    # ── Class constants ────────────────────────────────────────────────────
    _DEFAULT_WATERMARK: str = "1970-01-01 00:00:00"
    _TABLE_NAME: str = "core_transactions"
    _HIGH_VALUE_THRESHOLD: int = 10_000     # AED threshold for HIGH_VALUE category
    _JDBC_PARTITION_COUNT: int = 20         # parallel JDBC partitions for extraction
    _JDBC_FETCH_SIZE: int = 5_000          # rows per JDBC batch fetch

    def __init__(
        self,
        spark: SparkSession,
        config: Dict[str, str],
    ) -> None:
        """
        Initialise the ETL pipeline with Spark session and configuration.

        Args:
            spark  (SparkSession)   : Active SparkSession; must already be created
                                      by the caller (e.g. Databricks cluster).
            config (Dict[str, str]) : Pipeline configuration dictionary.
                                      Required keys:
                                        oracle_jdbc_url, db_user, db_password,
                                        watermark_path, silver_path,
                                        quarantine_path, branch_table.

        Raises:
            ValueError: If any required configuration key is missing.
        """
        self._validate_config(config)
        self.spark: SparkSession = spark
        self.config: Dict[str, str] = config
        self.run_date: str = datetime.today().strftime("%Y-%m-%d")
        self._jdbc_props: Optional[Dict[str, str]] = None
        self._setup_logging()

    # ════════════════════════════════════════════════════════════════════
    # SETUP METHODS
    # ════════════════════════════════════════════════════════════════════

    def _setup_logging(self) -> None:
        """
        Configure module-level logger with a standard formatter.

        Sets INFO level by default; caller may override via logging.basicConfig
        before instantiating this class.
        """
        if not logger.handlers:
            handler = logging.StreamHandler()
            fmt = logging.Formatter(
                "%(asctime)s [%(levelname)s] %(name)s: %(message)s",
                datefmt="%Y-%m-%d %H:%M:%S",
            )
            handler.setFormatter(fmt)
            logger.addHandler(handler)
            logger.setLevel(logging.INFO)

    def _setup_spark_config(self) -> None:
        """
        Apply performance-optimised Spark configuration for this pipeline.

        Enables Adaptive Query Execution (AQE) for runtime re-planning,
        coalesces small shuffle partitions automatically, and activates
        skew-join splitting to prevent single-executor bottlenecks.
        All settings are idempotent and safe to call multiple times.
        """
        conf_pairs: List[Tuple[str, str]] = [
            ("spark.sql.adaptive.enabled",                       "true"),
            ("spark.sql.adaptive.coalescePartitions.enabled",     "true"),
            ("spark.sql.adaptive.skewJoin.enabled",               "true"),
            ("spark.sql.adaptive.advisoryPartitionSizeInBytes",    "134217728"),  # 128 MB
            ("spark.sql.shuffle.partitions",                       "400"),
            ("spark.sql.autoBroadcastJoinThreshold",               "20971520"),   # 20 MB
        ]
        for key, value in conf_pairs:
            self.spark.conf.set(key, value)
        logger.info("Spark configuration applied successfully.")

    def _setup_jdbc_properties(self) -> Dict[str, str]:
        """
        Build and cache reusable JDBC connection properties for Oracle source.

        Properties are initialised once and reused across all JDBC calls to
        avoid repeated dictionary construction in hot paths.

        Returns:
            Dict[str, str]: JDBC driver properties including parallelism settings.
        """
        if self._jdbc_props is None:
            self._jdbc_props = {
                "url":             self.config["oracle_jdbc_url"],
                "driver":          "oracle.jdbc.OracleDriver",
                "user":            self.config["db_user"],
                "password":        self.config["db_password"],
                "fetchsize":       str(self._JDBC_FETCH_SIZE),
                "numPartitions":   str(self._JDBC_PARTITION_COUNT),
                "partitionColumn": "TXN_ID",
                "lowerBound":      "1",
                "upperBound":      "1000000000",
            }
        return self._jdbc_props

    def _validate_config(self, config: Dict[str, str]) -> None:
        """
        Assert that all required configuration keys are present.

        Args:
            config (Dict[str, str]): Configuration dictionary to validate.

        Raises:
            ValueError: If any required key is absent from config.
        """
        required_keys: List[str] = [
            "oracle_jdbc_url", "db_user", "db_password",
            "watermark_path", "silver_path", "quarantine_path",
            "branch_table",
        ]
        missing = [k for k in required_keys if k not in config]
        if missing:
            raise ValueError(
                f"Missing required config keys: {missing}"
            )

    # ════════════════════════════════════════════════════════════════════
    # LOAD METHODS
    # ════════════════════════════════════════════════════════════════════

    def _load_watermark(self) -> str:
        """
        Read the last successfully processed timestamp from the metadata table.

        Uses the pipeline's table name as the filter key so multiple pipelines
        can share a single watermark Delta table without collision.

        Returns:
            str: ISO timestamp string of the last processed batch.
                 Returns _DEFAULT_WATERMARK ('1970-01-01 00:00:00') if no
                 prior run is recorded (first-time execution).
        """
        try:
            rows = (
                self.spark.read.format("delta")
                .load(self.config["watermark_path"])
                .filter(col("table_name") == self._TABLE_NAME)
                .select("last_processed_ts")
                .collect()
            )
            watermark = str(rows[0][0]) if rows else self._DEFAULT_WATERMARK
        except Exception:  # noqa: BLE001 — tolerate missing table on first run
            watermark = self._DEFAULT_WATERMARK

        logger.info("Loaded watermark: %s", watermark)
        return watermark

    def _load_source_data(self, last_wm: str) -> DataFrame:
        """
        Extract incremental transaction records from Oracle core banking via JDBC.

        Applies a server-side WHERE filter on LAST_UPDATED to limit extraction
        to rows modified after the last successful pipeline run. Uses parallel
        JDBC partitioning to prevent single-thread throughput bottlenecks.

        Args:
            last_wm (str): Watermark timestamp. Only rows with
                           LAST_UPDATED > last_wm are extracted.

        Returns:
            DataFrame: Raw incremental records from Oracle; schema reflects
                       the source table structure without modifications.

        Raises:
            RuntimeError: If the JDBC connection fails or Oracle is unavailable.
        """
        jdbc_props = self._setup_jdbc_properties()
        # Server-side push-down: Oracle filters rows before network transfer
        push_down_query = (
            f"(SELECT * FROM CORE_TRANSACTIONS WHERE LAST_UPDATED > "
            f"TO_TIMESTAMP('{last_wm}', 'YYYY-MM-DD HH24:MI:SS')) T"
        )
        try:
            raw_df = (
                self.spark.read.format("jdbc")
                .option("dbtable", push_down_query)
                .options(**jdbc_props)
                .load()
            )
            logger.info("Extracted %d rows from Oracle.", raw_df.count())
            return raw_df
        except Exception as exc:
            raise RuntimeError(
                f"JDBC extraction failed for watermark={last_wm}: {exc}"
            ) from exc

    def _load_branch_reference(self) -> DataFrame:
        """
        Load the branch master dimension table for enrichment joins.

        The branch table is small (~500 rows) and changes infrequently.
        Callers should apply broadcast() when joining to avoid shuffle.

        Returns:
            DataFrame: Branch master with columns (branch_id, branch_name, region).
        """
        branch_df = (
            self.spark.table(self.config["branch_table"])
            .select("branch_id", "branch_name", "region")
        )
        logger.info("Loaded %d branch reference rows.", branch_df.count())
        return branch_df

    # ════════════════════════════════════════════════════════════════════
    # PROCESS METHODS
    # ════════════════════════════════════════════════════════════════════

    def _process_validate(
        self,
        df: DataFrame,
    ) -> Dict[str, DataFrame]:
        """
        Apply data quality rules and route records to valid or quarantine sets.

        Each failed record is tagged with a pipe-separated error string
        listing every violated rule. Records violating one or more rules
        are routed to quarantine rather than silently dropped, enabling
        downstream human review and reprocessing.

        Args:
            df (DataFrame): Raw Bronze-layer records to validate.

        Returns:
            Dict[str, DataFrame]: Dictionary with two keys:
                'valid'      — Records passing all quality rules.
                'quarantine' — Records failing one or more rules, with
                               an additional 'error_codes' string column.
        """
        # Tag each record with an array of triggered error codes
        tagged_df = df.withColumn(
            "_error_array",
            array(
                when(col("TXN_ID").isNull(),        lit("NULL_TXN_ID")),
                when(col("CUSTOMER_ID").isNull(),  lit("NULL_CUSTOMER_ID")),
                when(col("AMOUNT") <= 0,            lit("NON_POSITIVE_AMOUNT")),
                when(col("BRANCH_ID").isNull(),    lit("NULL_BRANCH_ID")),
                when(
                    ~col("STATUS").isin(
                        "COMPLETED", "PENDING", "FAILED"
                    ),
                    lit("INVALID_STATUS"),
                ),
            ),
        )

        # Filter out null entries from the error array, then flatten to string
        active_errors_df = tagged_df.withColumn(
            "_active_errors",
            array_filter("_error_array", lambda x: x.isNotNull()),
        ).withColumn(
            "error_codes",
            concat_ws("|", col("_active_errors")),
        ).drop("_error_array", "_active_errors")

        valid_df = (
            active_errors_df
            .filter(col("error_codes") == "")
            .drop("error_codes")
        )
        quarantine_df = active_errors_df.filter(
            col("error_codes") != ""
        )
        logger.info(
            "Validation: %d valid, %d quarantined.",
            valid_df.count(),
            quarantine_df.count(),
        )
        return {"valid": valid_df, "quarantine": quarantine_df}

    def _process_deduplicate(self, df: DataFrame) -> DataFrame:
        """
        Remove duplicate records keeping the most recently ingested version.

        Uses a row_number window function partitioned by business key TXN_ID
        and ordered by _load_ts descending. This deterministically retains
        the latest record when the source retransmits duplicates.

        Args:
            df (DataFrame): Validated records potentially containing duplicates.

        Returns:
            DataFrame: Deduplicated records; exactly one row per TXN_ID.
        """
        dedup_window = (
            Window.partitionBy("TXN_ID")
            .orderBy(col("_load_ts").desc())
        )
        deduplicated_df = (
            df.withColumn("_rn", row_number().over(dedup_window))
            .filter(col("_rn") == 1)
            .drop("_rn")
        )
        logger.info(
            "Deduplication: %d → %d rows.",
            df.count(),
            deduplicated_df.count(),
        )
        return deduplicated_df

    def _process_transform(self, df: DataFrame) -> DataFrame:
        """
        Apply business transformation rules to produce Silver-ready records.

        Transformations applied (in order):
            1. txn_category  — classifies AMOUNT into HIGH_VALUE / STANDARD
            2. _load_ts      — ingestion audit timestamp (current UTC time)
            3. _pipeline_run — run date for partitioning and lineage tracing

        Args:
            df (DataFrame): Deduplicated validated records.

        Returns:
            DataFrame: Transformed records with additional computed columns.
        """
        transformed_df = (
            df
            # Business rule: classify transaction by value band
            .withColumn(
                "txn_category",
                when(
                    col("AMOUNT") >= self._HIGH_VALUE_THRESHOLD,
                    lit("HIGH_VALUE"),
                ).otherwise(lit("STANDARD")),
            )
            # Audit columns — always add last to avoid impacting earlier logic
            .withColumn("_load_ts", current_timestamp())
            .withColumn("_pipeline_run", lit(self.run_date))
        )
        return transformed_df

    # ════════════════════════════════════════════════════════════════════
    # COMBINE + FINALIZE METHOD
    # ════════════════════════════════════════════════════════════════════

    def _combine_and_finalize_data(
        self,
        txn_df: DataFrame,
        branch_df: DataFrame,
    ) -> DataFrame:
        """
        Enrich transformed transactions with branch metadata and finalise schema.

        Applies a broadcast join on branch_id (branch table is small — ~500 rows)
        to eliminate shuffle and then selects the canonical Silver output schema.
        This method is the single point of schema definition for the Silver table.

        Args:
            txn_df    (DataFrame): Transformed transaction records.
            branch_df (DataFrame): Branch master dimension (branch_id,
                                   branch_name, region).

        Returns:
            DataFrame: Fully enriched records conforming to the Silver schema.
        """
        from pyspark.sql.functions import broadcast  # local import avoids circular import risk

        enriched_df = txn_df.join(
            broadcast(branch_df),
            on="BRANCH_ID",
            how="left",
        )
        # Canonical Silver schema — explicit column selection prevents SELECT *
        final_df = enriched_df.select(
            col("TXN_ID").alias("txn_id"),
            col("CUSTOMER_ID").alias("customer_id"),
            col("AMOUNT").alias("amount"),
            col("STATUS").alias("txn_status"),
            col("TXN_DATE").alias("txn_date"),
            col("txn_category"),
            col("branch_name"),
            col("region"),
            col("_load_ts"),
            col("_pipeline_run"),
        )
        return final_df

    # ════════════════════════════════════════════════════════════════════
    # WRITE / OUTPUT METHODS
    # ════════════════════════════════════════════════════════════════════

    def _write_output(
        self,
        valid_df: DataFrame,
        quarantine_df: DataFrame,
    ) -> None:
        """
        Persist the valid Silver records and quarantine records to Delta Lake.

        Both writes use append mode to preserve full history. The quarantine
        table includes the error_codes column to enable targeted reprocessing.

        Args:
            valid_df      (DataFrame): Silver-ready records to append to the
                                       main transaction Delta table.
            quarantine_df (DataFrame): Records that failed validation, written
                                       to the quarantine Delta path.

        Raises:
            RuntimeError: If either write operation fails.
        """
        try:
            valid_df.write.format("delta").mode("append").save(
                self.config["silver_path"]
            )
            logger.info("Wrote %d rows to Silver.", valid_df.count())
        except Exception as exc:
            raise RuntimeError(
                f"Silver write failed: {exc}"
            ) from exc

        if quarantine_df.count() > 0:
            try:
                quarantine_df.write.format("delta").mode("append").save(
                    self.config["quarantine_path"]
                )
                logger.info(
                    "Wrote %d rows to quarantine.",
                    quarantine_df.count(),
                )
            except Exception as exc:
                raise RuntimeError(
                    f"Quarantine write failed: {exc}"
                ) from exc

    def _update_watermark(self, new_watermark: str) -> None:
        """
        Persist the new high-watermark back to the shared metadata Delta table.

        Called only after a successful Silver write so that a failed run leaves
        the watermark unchanged, enabling safe re-execution from the last known
        good state (exactly-once guarantee).

        Args:
            new_watermark (str): The maximum LAST_UPDATED value from the
                                 records successfully written in this run.
        """
        wm_df = self.spark.createDataFrame(
            [(self._TABLE_NAME, new_watermark, self.run_date)],
            schema="table_name STRING, last_processed_ts STRING, updated_date STRING",
        )
        wm_df.write.format("delta").mode("append").save(
            self.config["watermark_path"]
        )
        logger.info("Watermark updated to: %s", new_watermark)

    # ════════════════════════════════════════════════════════════════════
    # ORCHESTRATOR — run_etl_pipeline()
    # ════════════════════════════════════════════════════════════════════

    def run_etl_pipeline(self) -> None:
        """
        Master orchestrator — executes the full ETL pipeline end-to-end.

        Execution order:
            1.  Apply Spark performance configuration.
            2.  Read last-run watermark.
            3.  Extract incremental rows from Oracle.
            4.  Validate and split valid / quarantine records.
            5.  Deduplicate valid records.
            6.  Apply business transformations.
            7.  Load branch reference for enrichment.
            8.  Enrich and finalise Silver schema.
            9.  Write Silver + quarantine outputs.
            10. Update watermark only on full success.

        Raises:
            Exception: Re-raises any exception after structured logging so that
                       the ADF pipeline activity correctly registers FAILED
                       status and triggers retry or alert logic.
        """
        logger.info(
            "Starting CustomerTransactionETL for run_date=%s",
            self.run_date,
        )
        try:
            # ── 1. Configure Spark ──────────────────────────────────────────
            self._setup_spark_config()

            # ── 2. Load watermark ───────────────────────────────────────────
            last_wm = self._load_watermark()

            # ── 3. Extract source data ──────────────────────────────────────
            raw_df = self._load_source_data(last_wm)

            # ── 4. Validate: split valid vs quarantine ──────────────────────
            split: Dict[str, DataFrame] = self._process_validate(raw_df)

            # ── 5. Deduplicate valid records ────────────────────────────────
            deduped_df = self._process_deduplicate(split["valid"])

            # ── 6. Apply business transformations ───────────────────────────
            transformed_df = self._process_transform(deduped_df)

            # ── 7. Load branch reference ────────────────────────────────────
            branch_df = self._load_branch_reference()

            # ── 8. Enrich and finalise Silver schema ─────────────────────────
            final_df = self._combine_and_finalize_data(transformed_df, branch_df)

            # ── 9. Write outputs ────────────────────────────────────────────
            self._write_output(final_df, split["quarantine"])

            # ── 10. Update watermark (only on full success) ──────────────────
            new_wm = raw_df.agg(
                spark_max("LAST_UPDATED")
            ).collect()[0][0]
            if new_wm:
                self._update_watermark(str(new_wm))

            logger.info(
                "CustomerTransactionETL completed successfully for %s.",
                self.run_date,
            )
        except Exception as exc:
            logger.error(
                "CustomerTransactionETL FAILED for run_date=%s: %s",
                self.run_date,
                exc,
            )
            raise

class TestConfiguration:
    """Validate ETL initialisation and configuration behaviour."""

    def test_init_stores_config(
        self, etl: CustomerTransactionETL, config: dict
    ) -> None:
        """ETL instance must store the config dict passed at construction."""
        assert etl.config == config

    def test_missing_config_raises_value_error(
        self, spark: SparkSession
    ) -> None:
        """Instantiation with incomplete config must raise ValueError."""
        incomplete_config = {"oracle_jdbc_url": "jdbc:oracle:thin:@x:1521/X"}
        with pytest.raises(ValueError, match="Missing required config keys"):
            CustomerTransactionETL(spark, incomplete_config)

    def test_jdbc_properties_contain_required_keys(
        self, etl: CustomerTransactionETL
    ) -> None:
        """JDBC properties must contain all keys needed for parallel extraction."""
        props = etl._setup_jdbc_properties()
        for expected_key in ["url", "driver", "numPartitions", "partitionColumn"]:
            assert expected_key in props, f"Missing key: {expected_key}"

    def test_jdbc_properties_are_cached(
        self, etl: CustomerTransactionETL
    ) -> None:
        """Second call to _setup_jdbc_properties must return the cached object."""
        first_call  = etl._setup_jdbc_properties()
        second_call = etl._setup_jdbc_properties()
        assert first_call is second_call  # identical object — not a copy

    def test_run_date_format_is_valid(
        self, etl: CustomerTransactionETL

class TestLoading:
    """Validate watermark loading behaviour, including edge cases."""

    def test_watermark_returns_default_when_no_prior_run(
        self,
        etl: CustomerTransactionETL,
        spark: SparkSession,
    ) -> None:
        """
        _load_watermark must return the epoch default when the metadata
        table contains no rows for this pipeline's table_name.
        """
        empty_wm_df = spark.createDataFrame(
            [], schema="table_name STRING, last_processed_ts STRING"
        )
        with patch.object(
            spark.read.format("delta").__class__,
            "load",
            return_value=empty_wm_df,
        ):
            wm = etl._load_watermark()
        assert wm == CustomerTransactionETL._DEFAULT_WATERMARK

    def test_watermark_returns_stored_value(
        self,
        etl: CustomerTransactionETL,
        spark: SparkSession,
    ) -> None:
        """
        _load_watermark must return the stored timestamp when one exists
        for the pipeline's table_name in the metadata Delta table.
        """
        stored_wm = "2024-01-14 23:59:59"
        wm_df = spark.createDataFrame(
            [("core_transactions", stored_wm)],
            schema="table_name STRING, last_processed_ts STRING",
        )
        with patch.object(
            spark.read.format("delta").__class__, "load", return_value=wm_df
        ):
            wm = etl._load_watermark()
        assert wm == stored_wm

    def test_watermark_returns_default_on_read_exception(
        self,
        etl: CustomerTransactionETL,
    ) -> None:
        """
        _load_watermark must be tolerant: if the watermark Delta table does
        not yet exist (first run), it must return the default rather than raise.
        """
        with patch.object(etl.spark.read, "format", side_effect=Exception("No table")):
            wm = etl._load_watermark()
        assert wm == CustomerTransactionETL._DEFAULT_WATERMARK

class TestBusinessLogic:
    """
    Validate all business rules implemented in _process_validate,
    _process_transform, and _process_deduplicate.
    """

    def test_validate_null_txn_id_is_quarantined(
        self,
        etl: CustomerTransactionETL,
        sample_transactions,
    ) -> None:
        """Records with NULL TXN_ID must appear in quarantine with NULL_TXN_ID code."""
        result = etl._process_validate(sample_transactions)
        quarantine_errors = (
            result["quarantine"]
            .filter("error_codes LIKE '%NULL_TXN_ID%'")
        )
        assert quarantine_errors.count() == 1

    def test_validate_zero_amount_is_quarantined(
        self,
        etl: CustomerTransactionETL,
        sample_transactions,
    ) -> None:
        """Records with amount == 0 must be quarantined with NON_POSITIVE_AMOUNT."""
        result = etl._process_validate(sample_transactions)
        zero_amount = (
            result["quarantine"]
            .filter("error_codes LIKE '%NON_POSITIVE_AMOUNT%' AND TXN_ID = 'T004'")
        )
        assert zero_amount.count() == 1

    def test_validate_negative_amount_is_quarantined(
        self,
        etl: CustomerTransactionETL,
        sample_transactions,
    ) -> None:
        """Records with amount < 0 must be quarantined."""
        result = etl._process_validate(sample_transactions)
        negative = (
            result["quarantine"]
            .filter("TXN_ID = 'T005'")
        )
        assert negative.count() == 1

    def test_validate_invalid_status_is_quarantined(
        self,
        etl: CustomerTransactionETL,
        sample_transactions,
    ) -> None:
        """Records with STATUS not in (COMPLETED, PENDING, FAILED) must be quarantined."""
        result = etl._process_validate(sample_transactions)
        invalid_status = (
            result["quarantine"]
            .filter("TXN_ID = 'T006'")
        )
        assert invalid_status.count() == 1

    def test_validate_fully_valid_record_passes(
        self,
        etl: CustomerTransactionETL,
        sample_transactions,
    ) -> None:
        """
        Fully valid records must appear in the valid set and must NOT
        have an error_codes column (it is dropped before returning valid set).
        """
        result = etl._process_validate(sample_transactions)
        valid_df = result["valid"]
        assert valid_df.count() == 2                        # T001 and T002
        assert "error_codes" not in valid_df.columns      # column must be dropped

    def test_transform_high_value_category_applied(
        self,
        etl: CustomerTransactionETL,
        sample_transactions,
    ) -> None:
        """Records with AMOUNT >= 10000 must receive txn_category = 'HIGH_VALUE'."""
        valid_df = etl._process_validate(sample_transactions)["valid"]
        transformed = etl._process_transform(valid_df)
        high_value = transformed.filter(
            (transformed.AMOUNT >= Decimal("10000"))
            & (transformed.txn_category == "HIGH_VALUE")
        )
        assert high_value.count() == 1  # only T001 is HIGH_VALUE

    def test_transform_standard_category_applied(
        self,
        etl: CustomerTransactionETL,
        sample_transactions,
    ) -> None:
        """Records with AMOUNT < 10000 must receive txn_category = 'STANDARD'."""
        valid_df = etl._process_validate(sample_transactions)["valid"]
        transformed = etl._process_transform(valid_df)
        standard = transformed.filter(
            (transformed.AMOUNT < Decimal("10000"))
            & (transformed.txn_category == "STANDARD")
        )
        assert standard.count() == 1  # only T002 is STANDARD

    def test_transform_boundary_exactly_10000_is_high_value(
        self,
        etl: CustomerTransactionETL,
        spark: SparkSession,
    ) -> None:
        """
        Boundary test: AMOUNT exactly equal to the threshold (10000.00)
        must be classified as HIGH_VALUE, not STANDARD.
        """
        schema = StructType([
            StructField("TXN_ID", StringType()),
            StructField("CUSTOMER_ID", StringType()),
            StructField("BRANCH_ID", StringType()),
            StructField("AMOUNT", DecimalType(18, 2)),
            StructField("STATUS", StringType()),
            StructField("TXN_DATE", DateType()),
        ])
        boundary_data = [
            ("TB01", "C99", "B01", Decimal("10000.00"), "COMPLETED", date(2024, 1, 1))
        ]
        boundary_df = spark.createDataFrame(boundary_data, schema)
        # Add dummy _load_ts column so transform works without prior dedup step
        from pyspark.sql.functions import current_timestamp
        boundary_df = boundary_df.withColumn("_load_ts", current_timestamp())
        transformed = etl._process_transform(boundary_df)
        assert transformed.first()["txn_category"] == "HIGH_VALUE"

    def test_transform_adds_load_timestamp_column(
        self,
        etl: CustomerTransactionETL,
        sample_transactions,
    ) -> None:
        """_load_ts column must be present after _process_transform."""
        valid_df = etl._process_validate(sample_transactions)["valid"]
        transformed = etl._process_transform(valid_df)
        assert "_load_ts" in transformed.columns

    def test_deduplicate_keeps_one_per_txn_id(
        self,
        etl: CustomerTransactionETL,
        spark: SparkSession,
    ) -> None:
        """
        When two records share the same TXN_ID, _process_deduplicate must
        retain exactly one row (the latest by _load_ts).
        """
        from pyspark.sql.functions import current_timestamp, lit
        dup_schema = StructType([
            StructField("TXN_ID", StringType()),
            StructField("AMOUNT", DecimalType(18, 2)),
        ])
        dup_data = [
            ("DUP001", Decimal("100.00")),
            ("DUP001", Decimal("100.00")),  # exact duplicate
            ("UNQ001", Decimal("200.00")),
        ]
        dup_df = (
            spark.createDataFrame(dup_data, dup_schema)
            .withColumn("_load_ts", current_timestamp())
        )
        deduped = etl._process_deduplicate(dup_df)
        assert deduped.count() == 2  # DUP001 collapsed to 1, UNQ001 retained

class TestIntegration:
    """
    Validate end-to-end pipeline wiring using mocked external dependencies.
    Tests here confirm that the orchestrator correctly calls sub-methods
    and that data flows through the full pipeline correctly.
    """

    @patch("transaction_etl.CustomerTransactionETL._load_watermark")
    @patch("transaction_etl.CustomerTransactionETL._load_source_data")
    @patch("transaction_etl.CustomerTransactionETL._load_branch_reference")
    @patch("transaction_etl.CustomerTransactionETL._write_output")
    @patch("transaction_etl.CustomerTransactionETL._update_watermark")
    def test_pipeline_calls_watermark_before_extracting_data(
        self,
        mock_update_wm,
        mock_write,
        mock_branch,
        mock_load_data,
        mock_load_wm,
        etl: CustomerTransactionETL,
        sample_transactions,
        spark: SparkSession,
    ) -> None:
        """
        run_etl_pipeline must invoke _load_watermark before _load_source_data.
        The watermark value must gate what data is extracted — this test confirms
        the call sequence is correct.
        """
        mock_load_wm.return_value = "2024-01-01 00:00:00"
        mock_load_data.return_value = sample_transactions
        mock_branch.return_value = spark.createDataFrame(
            [("B01", "Main Branch", "Dubai"),
             ("B02", "Branch 2",    "Abu Dhabi")],
            schema="branch_id STRING, branch_name STRING, region STRING",
        )
        mock_write.return_value = None
        mock_update_wm.return_value = None

        etl.run_etl_pipeline()

        mock_load_wm.assert_called_once()
        # _load_source_data must be called with the watermark value
        mock_load_data.assert_called_once_with("2024-01-01 00:00:00")

    def test_quarantine_count_matches_invalid_input_records(
        self,
        etl: CustomerTransactionETL,
        sample_transactions,
    ) -> None:
        """
        The number of records routed to quarantine must exactly match
        the number of records in the input that violate at least one rule.
        Sample fixture has 4 invalid records (T003 NULL, T004 zero, T005 neg, T006 bad status).
        """
        result = etl._process_validate(sample_transactions)
        assert result["quarantine"].count() == 4
        assert result["valid"].count() == 2
        # Confirm counts are mutually exclusive and add up to total
        total = result["valid"].count() + result["quarantine"].count()
        assert total == sample_transactions.count()
